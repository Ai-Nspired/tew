      for (var i = 0; i < all.length; i++) setChip(all[i], false);
      setChip(c, !wasOn);
    } else {
      setChip(c, c.getAttribute('data-on') !== 'true');
    }
    if (canSave) { clearTimeout(timer); timer = setTimeout(save, 400); }
  });

    var locked = doc.getAttribute('contenteditable') === 'false';
    doc.setAttribute('contenteditable', locked ? 'true' : 'false');
    editBtn.textContent = locked ? '🔒 Lock' : '✏️ Edit';
  });

    var out = '<!DOCTYPE html>\n' + document.documentElement.outerHTML;

    a.href = URL.createObjectURL(new Blob([out], {type:'text/html'}));
    a.download = 'life-estate-grant-deed-' + new Date().toISOString().slice(0,10) + '.html';
    document.body.appendChild(a); a.click(); a.remove();
    setTimeout(function(){ URL.revokeObjectURL(a.href); }, 1000);
  });

    if (confirm('Clear all edits and restore the blank template? This cannot be undone.')) {
      try { localStorage.removeItem(KEY); } catch (e) {}
      location.reload();
    }
  });
